<script>
(function () {
  const sections = Array.from(document.querySelectorAll('section[data-view]'));
  const label = document.getElementById('view');
  function show() {
    const wanted = location.hash.slice(1);
    const target = sections.find(function (s) { return s.dataset.view === wanted; }) || sections[0];
    sections.forEach(function (s) { s.hidden = s !== target; });
    label.textContent = target.dataset.title;
    window.scrollTo(0, 0);
    document.body.dataset.view = target.dataset.view;
  }
  window.addEventListener('hashchange', show);
  show();
  document.body.dataset.ready = 'true';
})();
</script>
