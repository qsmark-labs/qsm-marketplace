<script>
(function () {
  const viewer = new BpmnJS({ container: '#canvas' });
  const crumbs = [];
  const nav = document.getElementById('crumbs');
  const details = document.getElementById('details');
  const STEP = /Task$|^bpmn:(CallActivity|SubProcess)$/;
  const MIN_ZOOM = 0.9;

  function processOf(diagram) {
    const e = diagram.plane.bpmnElement;
    if (e.$type === 'bpmn:Process') return e;
    const p = (e.participants || [])[0];
    return p && p.processRef;
  }
  function diagramOf(processId) {
    return viewer.getDefinitions().diagrams.find(function (d) { const p = processOf(d); return p && p.id === processId; });
  }
  function nameOf(diagram) { const p = processOf(diagram); return (p && (p.name || p.id)) || diagram.id; }

  function drawCrumbs() {
    nav.replaceChildren();
    crumbs.forEach(function (c, i) {
      if (i > 0) { const s = document.createElement('span'); s.className = 'sep'; s.textContent = '›'; nav.append(s); }
      const b = document.createElement('button');
      b.type = 'button';
      b.textContent = c.label;
      if (i === crumbs.length - 1) b.setAttribute('aria-current', 'page');
      else b.addEventListener('click', function () { crumbs.length = i + 1; show(c.diagram); });
      nav.append(b);
    });
  }

  function show(diagram) {
    return viewer.open(diagram).then(function () {
      // a wide process fitted to the window shrinks its labels below reading size: below MIN_ZOOM
      // it opens at MIN_ZOOM from its top left, and the reader drags to see the rest
      const canvas = viewer.get('canvas');
      canvas.zoom('fit-viewport');
      if (canvas.zoom() < MIN_ZOOM) {
        const box = canvas.viewbox();
        canvas.viewbox({ x: box.inner.x - 20, y: box.inner.y - 20, width: box.outer.width / MIN_ZOOM, height: box.outer.height / MIN_ZOOM });
      }
      drawCrumbs();
      details.replaceChildren();
      const p = document.createElement('p'); p.className = 'hint';
      p.textContent = 'Click a step to see its lane, its name and its notes. Drag to move the diagram, Ctrl and the wheel to zoom.';
      details.append(p);
    });
  }

  function laneOf(element) {
    const lanes = viewer.get('elementRegistry').filter(function (e) { return e.type === 'bpmn:Lane'; });
    const lane = lanes.find(function (l) { return (l.businessObject.flowNodeRef || []).indexOf(element.businessObject) >= 0; });
    return lane ? lane.businessObject.name : '';
  }

  function row(dl, term, value) {
    const dt = document.createElement('dt'); dt.textContent = term;
    const dd = document.createElement('dd');
    if (value instanceof Node) dd.append(value); else dd.textContent = value;
    dl.append(dt, dd);
  }

  function describe(element) {
    const bo = element.businessObject;
    const dl = document.createElement('dl');
    row(dl, 'Lane', laneOf(element) || '(none)');
    row(dl, 'Name', bo.name || bo.id);
    const notes = (bo.documentation || []).filter(function (d) { return d.textFormat === 'text/x-gherkin'; });
    if (notes.length === 0) {
      row(dl, 'Scenarios', 'none written yet');
    } else {
      notes.forEach(function (d) { const pre = document.createElement('pre'); pre.textContent = d.text || ''; row(dl, 'Scenarios', pre); });
    }
    if (bo.$type === 'bpmn:CallActivity') row(dl, 'Calls', bo.calledElement || '');
    details.replaceChildren(dl);
  }

  viewer.importXML(document.getElementById('graph').textContent).then(function () {
    const main = viewer.getDefinitions().diagrams[0];
    crumbs.push({ label: nameOf(main), diagram: main });
    viewer.get('eventBus').on('element.click', function (ev) {
      const bo = ev.element.businessObject;
      if (!STEP.test(ev.element.type)) return;
      describe(ev.element);
      if (bo.$type !== 'bpmn:CallActivity') return;
      const d = diagramOf(bo.calledElement);
      if (!d) return;
      crumbs.push({ label: nameOf(d), diagram: d });
      show(d);
    });
    return show(main);
  }).then(function () { document.body.dataset.ready = 'true'; }, function (err) {
    details.textContent = 'The graph could not be drawn: ' + err.message;
    document.body.dataset.ready = 'failed';
  });
})();
</script>
